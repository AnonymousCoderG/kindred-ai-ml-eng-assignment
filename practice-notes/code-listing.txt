    28	class SimulatedCrash(Exception):
    29	    """Lets a caller reproduce a process crash between the charge succeeding
    30	    and the session/booking being persisted — the same ambiguous-outcome
    31	    scenario as a real payment-provider timeout, but deterministic to trigger."""

    34	def advance_turn(session: ConversationSession, intent: str, payload: dict, db: DbSession) -> dict:
    35	    if intent == "refund_dispute":
    36	        session.status = "escalated"
    37	        db.commit()
    38	        return {"status": "escalated"}
    39	
    40	    if intent == "book":
    41	        party_size = payload.get("party_size")
    42	        if party_size is not None:
    43	            session.party_size = party_size
    44	        if session.party_size is not None:
    45	            session.awaiting_confirmation = True
    46	        db.commit()
    47	        return {
    48	            "status": "awaiting_confirmation" if session.awaiting_confirmation else "need_party_size",
    49	            "party_size": session.party_size,
    50	        }
    51	
    52	    if intent == "affirm" and session.awaiting_confirmation:
    53	        amount_cents = payload.get("amount_cents", 0)
    54	
    55	        # No check here for a prior successful charge on this session, and
    56	        # nothing is persisted before the charge happens.
    57	        result = payment_mock_client.charge(amount_cents)
    58	
    59	        if payload.get("simulate_crash"):
    60	            raise SimulatedCrash("process died after the charge, before the session/booking were saved")
    61	
    62	        booking = Booking(
    63	            member_id=session.member_id,
    64	            club_id=session.club_id,
    65	            description="Session-confirmed booking",
    66	            amount_cents=amount_cents,
    67	            status="confirmed" if result.status == "succeeded" else "pending",
    68	        )
    69	        db.add(booking)
    70	        db.flush()
    71	        db.add(PaymentAttempt(booking_id=booking.id, amount_cents=amount_cents, status=result.status))
    72	        session.booking_id = booking.id
    73	        session.status = "confirmed"
    74	        session.awaiting_confirmation = False
    75	        db.commit()
    76	        return {"status": "confirmed", "booking_id": booking.id}
    77	
    78	    return {"status": session.status}

    93	def cosine_similarity(a: list[float], b: list[float]) -> float:
    94	    dot = sum(x * y for x, y in zip(a, b))
    95	    norm_a = math.sqrt(sum(x * x for x in a))
    96	    norm_b = math.sqrt(sum(y * y for y in b))
    97	    if norm_a == 0 or norm_b == 0:
    98	        return 0.0
    99	    return dot / (norm_a * norm_b)

   102	def rank_candidates(member: Member, reason: str, db: Session) -> list[dict]:
   103	    candidates = (
   104	        db.query(Member)
   105	        .filter(Member.club_id == member.club_id, Member.id != member.id)
   106	        .all()
   107	    )
   108	
   109	    results = []
   110	    for candidate in candidates:
   111	        # Recomputes the requesting member's own embedding on every iteration
   112	        # via embedding_client.embed(), instead of once before the loop.
   113	        query_vec = embedding_client.embed(build_member_profile_text(member, db))
   114	        candidate_vec = candidate.profile_embedding
   115	        if candidate_vec is None:
   116	            continue
   117	        score = cosine_similarity(query_vec, [float(v) for v in candidate_vec])
   118	        results.append({"member_id": candidate.id, "name": candidate.name, "score": round(float(score), 4)})
   119	
   120	    results.sort(key=lambda r: -r["score"])
   121	    return results

   231	class PaymentTimeoutError(Exception):
   232	    """Raised when the (mock) payment provider times out with an ambiguous
   233	    response — the caller does not know whether the charge went through."""

   238	    status: str  # "succeeded" | "failed"

   241	class PaymentMockClient:
   242	    def __init__(self) -> None:
   243	        # Every attempted charge, in order — a stand-in for the provider's own
   244	        # ledger, useful for proving a bug caused two real-world charges.
   245	        self.charge_log: list[int] = []
   246	
   247	    def charge(self, amount_cents: int) -> ChargeResult:
   248	        self.charge_log.append(amount_cents)
   249	        if amount_cents == settings.payment_timeout_trigger_cents:
   250	            raise PaymentTimeoutError("provider timed out")
   251	        if amount_cents <= 0:
   252	            return ChargeResult(status="failed")
   253	        return ChargeResult(status="succeeded")
